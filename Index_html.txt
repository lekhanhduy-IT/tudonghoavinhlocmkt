var adminTabs = [
        { id: "YTUONG", name: "Ý TƯỞNG", icon: "fa-lightbulb" },
        { id: "KEHOACH", name: "KẾ HOẠCH", icon: "fa-calendar-alt" },

        { id: "HINHANH", name: "HÌNH ANH", icon: "fa-images" },
        { id: "CAIDAT", name: "CÀI ĐẶT", icon: "fa-cogs", adminOnly: true }
      ];
      
      var userTabs = [
        { id: "USER_HOME", name: "TRANG CHỦ", icon: "fa-home" },
        { id: "USER_GIOITHIEU", name: "GIỚI THIỆU", icon: "fa-info-circle" },

      window.addEventListener('load', function() {
        renderNavigationTabs();
        // Kích hoạt nạp dữ liệu ngầm cực nhanh ngay khi tải xong khung ứng dụng
        if (typeof initSpreadsheetEngine === "function") {
          initSpreadsheetEngine(publicSpreadsheetId);
        }
      });
      
      function renderNavigationTabs() {
        var container = document.getElementById("dynamicNavTabs");
        container.innerHTML = "";
        var activeArr = (currentMode === 'ADMIN') ? adminTabs : userTabs;
        
        activeArr.forEach(function(tab) {
          var btn = document.createElement("div");
          btn.className = "nav-btn" + (tab.adminOnly ? " admin-only" : "");
          btn.id = "navBtn_" + tab.id;
          btn.innerHTML = "<i class='fas " + tab.icon + "'></i> " + tab.name;
          btn.onclick = function() { activateViewPanel(tab.id); };
          container.appendChild(btn);
        });
        
        // Kích hoạt tab đầu tiên làm mặc định
        activateViewPanel(activeArr[0].id);
      }
      
      function switchSystemMode(mode) {
        currentMode = mode;
        document.getElementById("modeAdminBtn").className = "mode-toggle" + (mode==='ADMIN'?' active':'');
        document.getElementById("modeUserBtn").className = "mode-toggle" + (mode==='USER'?' active':'');
        renderNavigationTabs();
      }
      
      function activateViewPanel(panelId) {
        var panels = document.querySelectorAll(".view-panel");
        panels.forEach(function(p) { p.classList.remove("active"); });
        
        var target = document.getElementById("panel_" + panelId);
        if (target) target.classList.add("active");
        
        var btns = document.querySelectorAll(".nav-btn");
        btns.forEach(function(b) { b.classList.remove("active"); });
        
        var activeBtn = document.getElementById("navBtn_" + panelId);
        if (activeBtn) activeBtn.classList.add("active");
      }
      
      // Hộp thoại tùy biến phẳng không sử dụng thành phần mặc định của trình duyệt
      function showSystemModal(title, contentHtml, buttons) {
        document.getElementById("popupTitle").innerText = title;
        document.getElementById("popupCustomContent").innerHTML = contentHtml;
        var btnRow = document.getElementById("popupButtonsRow");
        btnRow.innerHTML = "";
        
        buttons.forEach(function(b) {
          var button = document.createElement("button");
          button.className = "popup-btn " + (b.className || "btn-cancel");
          button.innerText = b.text;
          button.onclick = function() {
            b.callback();
            closeSystemModal();
          };
          btnRow.appendChild(button);
        });
        document.getElementById("systemPopupOverlay").style.display = "block";
      }
      
      function closeSystemModal() {
        document.getElementById("systemPopupOverlay").style.display = "none";
      }
      
      // Tìm kiếm thông minh toàn hệ thống
      function executeGlobalSearch() {
        var keyword = document.getElementById("globalSystemSearch").value.trim().toLowerCase();
        if (typeof performGridSearch === "function") {
          performGridSearch(keyword);
        }
      }
