  // Quản lý trạng thái bộ nhớ đệm ứng dụng (Client Session Memory Store)
  var rawSpreadsheetCache = {};
  var attributesCache = {};
  var systemTabOrder = [];
  var currentActiveTab = "";
  var activeSpreadsheetId = "";
  
  // Quản lý tọa độ vùng chọn dữ liệu dãy ô (Shift + Click Chọn Dải vùng rộng)
  var selectionAnchor = null; // {r: rowIdx, c: colIdx}
  var selectedRangeBounds = null; // {minR, maxR, minC, maxC}
  var isSelectingFullRows = false;
  var isSelectingFullCols = false;

  /**
   * Khởi tạo cơ chế nạp dữ liệu giao diện ngầm ngay khi load ứng dụng
   */
/**
   * Khởi tạo cơ chế nạp dữ liệu giao diện ngầm ngay khi load ứng dụng
   */
function initSpreadsheetEngine(spreadsheetId) {
    activeSpreadsheetId = spreadsheetId;
    
    // 1. LỤC TÌM TRONG TRÍ NHỚ CỦA TRÌNH DUYỆT (LOCAL STORAGE)
    var offlineCache = localStorage.getItem('MKT_OFFLINE_CACHE_' + spreadsheetId);

    if (offlineCache && offlineCache !== "undefined") {
        // CÓ DỮ LIỆU CŨ: Ráp giao diện ngay lập tức trong 0.1 giây
        var parsedData = JSON.parse(offlineCache);
        rawSpreadsheetCache = parsedData.sheets;
        attributesCache = parsedData.attributes;
        systemTabOrder = parsedData.tabOrder;
        
        var firstVisibleTab = systemTabOrder.find(function(tabName) {
            var attrs = attributesCache[tabName] || {};
            return attrs.hidden !== true;
        });
        currentActiveTab = firstVisibleTab || systemTabOrder[0];
        
        renderTabControls();
        buildSpreadsheetGrid();
        
        // 2. Kích hoạt Radar nạp ngầm để kiểm tra xem có ai đổi dữ liệu lúc mình đang tắt app không
        fetchFreshDataBackground(spreadsheetId);
    } else {
        // LẦN ĐẦU TIÊN MỞ APP: Bắt buộc phải chờ tải từ Google Sheets
        fetchFreshDataBackground(spreadsheetId);
    }

    setupGlobalMouseInterceptors();
    startQueueManager(); 
    startSpiderSyncEngine(); 
  }

  // HÀM TẢI DỮ LIỆU NGẦM VÀ LƯU VÀO TRÍ NHỚ TRÌNH DUYỆT
  function fetchFreshDataBackground(spreadsheetId) {
    google.script.run.withSuccessHandler(function(response) {
        if (response.success) {
            // LƯU TOÀN BỘ VÀO Ổ CỨNG TRÌNH DUYỆT ĐỂ LẦN SAU MỞ LÀ CÓ NGAY
            localStorage.setItem('MKT_OFFLINE_CACHE_' + spreadsheetId, JSON.stringify(response.data));
            
            rawSpreadsheetCache = response.data.sheets;
            attributesCache = response.data.attributes;
            systemTabOrder = response.data.tabOrder;
            
            if(!currentActiveTab) {
                var firstVisibleTab = systemTabOrder.find(function(tabName) {
                    var attrs = attributesCache[tabName] || {};
                    return attrs.hidden !== true;
                });
                currentActiveTab = firstVisibleTab || systemTabOrder[0];
            }
            
            // Vẽ lại bảng một cách êm ái
            renderTabControls();
            buildSpreadsheetGrid();
        }
    }).getInitialSystemData(spreadsheetId);
  }
// Bộ nhớ đệm hàng đợi
var actionQueue = [];
var isProcessingQueue = false;

// Kích hoạt vòng lặp xử lý ngầm (Chạy mỗi 2 giây)
// Kích hoạt vòng lặp xử lý ngầm (Chạy mỗi 2 giây)
function startQueueManager() {
  setInterval(function() {
    if (isProcessingQueue || actionQueue.length === 0) return;
    
    isProcessingQueue = true;
    var batchToProcess = actionQueue.splice(0, actionQueue.length); // Rút toàn bộ Queue ra
    
    google.script.run.withSuccessHandler(function(res) {
      isProcessingQueue = false; 
      
      // ⚡ CHÌA KHÓA 1: Bắt Sync Token từ Server trả về để Radar không tự tải lại dữ liệu do chính mình vừa sửa
      if (res && res.newSyncToken) {
          currentSyncToken = res.newSyncToken;
      }
      
    }).withFailureHandler(function(err) {
      console.warn("Mạng lỗi, đang lưu trữ thao tác chờ khôi phục...");
      actionQueue = batchToProcess.concat(actionQueue);
      isProcessingQueue = false;
    }).processActionBatchBackground(batchToProcess); 
    
  }, 2000);
}

// Sửa lại hàm cập nhật ô của bạn
function updateCellCacheAndSyncBackground(rIdx, cIdx, newValue) {
  // 1. Optimistic UI: Cập nhật giao diện lập tức (Code cũ bạn đã làm tốt phần này)
  rawSpreadsheetCache[currentActiveTab][rIdx][cIdx] = newValue;
  
  // 2. Không gọi API trực tiếp nữa, tống nó vào Queue
  actionQueue.push({
    action: 'SYNC_CELL',
    sheetId: activeSpreadsheetId,
    tab: currentActiveTab,
    row: rIdx + 1,
    col: cIdx + 1,
    val: newValue,
    timestamp: new Date().getTime()
  });
}

  /**
   * Kết xuất thanh tab chân trang và thiết lập sự kiện chuột phải tương tác tab
   */
  function renderTabControls() {
    var wrapper = document.getElementById("tabsListWrapper");
    wrapper.innerHTML = "";
    
    systemTabOrder.forEach(function(tabName) {
      // Bỏ qua nếu cấu hình ẩn trong tệp thuộc tính mở rộng
      var attrs = attributesCache[tabName] || {};
      if (attrs.hidden === true) return;
      
      var tabEl = document.createElement("div");
      tabEl.className = "tab-item-element" + (tabName === currentActiveTab ? " active" : "");
      tabEl.innerText = tabName;
      
      // Chuyển tab lập tức không cần đợi tải lại dữ liệu từ máy chủ mạng
      tabEl.onclick = function() {
        currentActiveTab = tabName;
        renderTabControls();
        buildSpreadsheetGrid();
      };
      
      // Gắn sự kiện chuột phải cấu hình thuộc tính tab ẩn/xóa/duplicate
      tabEl.oncontextmenu = function(e) {
        e.preventDefault();
        e.stopPropagation();
        openTabContextMenu(e, tabName);
      };
      
      wrapper.appendChild(tabEl);
    });
  }

  /**
   * Kỹ thuật xây dựng cấu trúc bảng Excel mượt mà tốc độ cao từ Cache
   */
/**
   * Kỹ thuật xây dựng cấu trúc bảng Excel mượt mà tốc độ cao từ Cache
   */
  function buildSpreadsheetGrid() {
    var dataMatrix = rawSpreadsheetCache[currentActiveTab] || [[""]];
    var attrs = attributesCache[currentActiveTab] || {};
    var columnTypes = attrs.columnTypes || {};
    var headEl = document.getElementById("mktTableHead");
    var bodyEl = document.getElementById("mktTableBody");
    headEl.innerHTML = ""; bodyEl.innerHTML = "";
    if (dataMatrix.length === 0) return;
    
    var colCount = dataMatrix[0].length;
    var frozenRows = attrs.frozenRows || 0;
    var frozenCols = attrs.frozenCols || 0;

    // Đoạn Tính Tọa độ Cố định Cột (Sticky offset)
    var colWidthsArr = [];
    var currentLeft = 50; 
    for (var c = 0; c < colCount; c++) {
      var w = (attrs.colWidths && attrs.colWidths[c]) ? parseInt(attrs.colWidths[c]) : 150;
      colWidthsArr[c] = { width: w, left: currentLeft };
      currentLeft += w;
    }

    // Đoạn Tính Tọa độ Cố định Hàng (Sticky offset)
    var rowHeightsArr = [];
    var currentTop = 21; 
    for(var r = 0; r < dataMatrix.length; r++) {
      var h = (attrs.rowHeights && attrs.rowHeights[r]) ? parseInt(attrs.rowHeights[r]) : 21;
      rowHeightsArr[r] = { height: h, top: currentTop };
      currentTop += h;
    }

    // 1. Tạo Hàng Tiêu Đề Cột A, B, C...
    var trHead = document.createElement("tr");
    var thIndex = document.createElement("th");
    thIndex.className = "col-index-header";
    thIndex.innerText = "#";
    thIndex.style.position = 'sticky'; thIndex.style.left = '0px'; thIndex.style.top = '0px'; thIndex.style.zIndex = '20';
    
    thIndex.onclick = function(e) {
      e.preventDefault();
      selectedRangeBounds = { minR: 0, maxR: dataMatrix.length - 1, minC: 0, maxC: colCount - 1 };

  function getColumnLetterName(colIdx) {
    var letter = "";
    while (colIdx >= 0) {
      letter = String.fromCharCode((colIdx % 26) + 65) + letter;
      colIdx = Math.floor(colIdx / 26) - 1;
    }
    return letter;
  }

  /**
   * Khôi phục trực quan màu sắc, định dạng đã lưu trữ từ bộ nhớ thuộc tính
   */
  function applyCellSavedAttributesInline(td, rIdx, cIdx, attrs) {
    var cellStyles = attrs.cellStyles || {};
    var key = rIdx + "_" + cIdx;
    var style = cellStyles[key] || {};
    
    // Thuộc tính dòng hoặc cột diện rộng
    var rowStyle = (attrs.rowStyles && attrs.rowStyles[rIdx]) ? attrs.rowStyles[rIdx] : {};
    var colStyle = (attrs.colStyles && attrs.colStyles[cIdx]) ? attrs.colStyles[cIdx] : {};
    
    var finalBg = style.bg || rowStyle.bg || colStyle.bg || "";
    var finalColor = style.color || rowStyle.color || colStyle.color || "";
    var finalBold = style.bold || rowStyle.bold || colStyle.bold || false;
    var finalItalic = style.italic || rowStyle.italic || colStyle.italic || false;
    var finalUnderline = style.underline || rowStyle.underline || colStyle.underline || false;
    var finalAlign = style.align || rowStyle.align || colStyle.align || "left";
    
    if (finalBg) td.style.backgroundColor = finalBg;
    if (finalColor) td.style.color = finalColor;
    if (finalBold) td.style.fontWeight = "bold";
    if (finalItalic) td.style.fontStyle = "italic";
    if (finalUnderline) td.style.textDecoration = "underline";
    td.style.textAlign = finalAlign;
  }

  /**
   * Kết xuất thành phần giao diện động tương thích chính xác loại dữ liệu đầu vào
   */
  function renderCellContentByType(td, rIdx, cIdx, val, type, attrs) {
    td.innerHTML = "";
    
    if (type === "checkbox") {
      var chk = document.createElement("input");
      chk.type = "checkbox";
      chk.checked = (val === true || val === "true" || val === 1 || val === "1");
      chk.style.margin = "8px";
      chk.onchange = function() {
        updateCellCacheAndSyncBackground(rIdx, cIdx, chk.checked);
      };
      td.appendChild(chk);
      return;
    }
    
    if (type === "dropdown") {
      var globalOptions = (attrs.dropdownChoices && attrs.dropdownChoices[cIdx]) ? attrs.dropdownChoices[cIdx] : ["Ý tưởng mới", "Đang xử lý", "Hoàn thành"];
      var currentSelections = String(val).split(",").map(function(s){return s.trim();}).filter(Boolean);

      // Tạo giao diện khung bo tròn
      var wrapper = document.createElement("div");
      wrapper.className = "cell-dropdown-ui";
      
      // Tạo chữ hiển thị bên trong
      var textSpan = document.createElement("div");
      textSpan.className = "dropdown-text";
      textSpan.innerText = currentSelections.length > 0 ? currentSelections.join(", ") : "Chọn mục...";
      if(currentSelections.length === 0) textSpan.style.color = "#999";
      
      // Tạo mũi tên ▼
      var arrow = document.createElement("div");
      arrow.className = "dropdown-arrow";
      arrow.innerHTML = "▼";
      
      wrapper.appendChild(textSpan);
      wrapper.appendChild(arrow);

      // Click chuột trái 1 LẦN để mở danh sách, không cần Double Click nữa
      wrapper.onclick = function(e) {
        e.stopPropagation();
        handleCellClickEngine(e, rIdx, cIdx); // Giữ nguyên hiệu ứng bôi đen ô
        openInlineDropdownList(e, rIdx, cIdx, currentSelections, globalOptions);
      };
      
      td.appendChild(wrapper);
      return;
    }
    
    if (type === "link") {
      var strVal = String(val);
      if (strVal.indexOf("http") === 0) {
        var a = document.createElement("a");
        a.href = strVal;
        a.target = "_blank";
        a.innerText = td.getAttribute("data-hyperlink-name") || "Xem liên kết";
        a.style.color = "var(--blue)";
        a.style.padding = "5px";
        a.style.display = "inline-block";
        td.appendChild(a);
      } else {
        var editor = createTextCellEditor(rIdx, cIdx, val);
        td.appendChild(editor);
      }
      return;
    }
    
    // Mặc định: loại dữ liệu TEXT (Xử lý RichText, Tự động chạy mã hiển thị nếu chứa thẻ HTML)
    var strValue = String(val);
    if (rIdx !== 0 && strValue.indexOf("<") === 0 && strValue.indexOf(">") !== -1) {
      // Tự động run mã và hiển thị trực tiếp lên giao diện ngay lập tức
      var htmlContainer = document.createElement("div");
      htmlContainer.style.padding = "5px";
      htmlContainer.innerHTML = strValue;
      td.appendChild(htmlContainer);
      
      td.ondblclick = function() {
        td.innerHTML = "";
        var editor = createTextCellEditor(rIdx, cIdx, strValue);
        td.appendChild(editor);
        editor.focus();
      };
    } else {
      var editor = createTextCellEditor(rIdx, cIdx, val);
      td.appendChild(editor);
    }
  }

function createTextCellEditor(rIdx, cIdx, val) {
    var div = document.createElement("div");
    div.className = "cell-rich-editor";
    div.contentEditable = "true";
    div.innerHTML = val;
    
    // Bắt sự kiện double-click để mở rộng diện tích ô
    div.ondblclick = function(e) {
      e.stopPropagation();
      if(this.classList.contains("cell-expanded")) {
        this.classList.remove("cell-expanded");
      } else {
        // Thu gọn các ô khác trước khi mở ô mới
        document.querySelectorAll(".cell-expanded").forEach(el => el.classList.remove("cell-expanded"));
        this.classList.add("cell-expanded");
      }
    };
    
    div.onblur = function() {
      this.classList.remove("cell-expanded"); // Tự thu gọn khi click ra ngoài
      updateCellCacheAndSyncBackground(rIdx, cIdx, div.innerHTML);
    };
    
    div.onkeydown = function(e) {
      if (e.key === "Tab") {
        var text = div.innerText.trim();
        if (text.indexOf("http") === 0) {
          e.preventDefault();
          updateCellCacheAndSyncBackground(rIdx, cIdx, text);
          buildSpreadsheetGrid();
        }
      }
    };
    return div;
  }

  /**
   * Cập nhật tức thời giao diện cục bộ và phát lệnh đồng bộ âm thầm về Server Apps Script
   */
  function updateCellCacheAndSyncBackground(rIdx, cIdx, newValue) {
    rawSpreadsheetCache[currentActiveTab][rIdx][cIdx] = newValue;
    // Chạy ngầm hoàn toàn không chặn UI người dùng
    google.script.run.syncCellValuesBackground(activeSpreadsheetId, currentActiveTab, rIdx + 1, cIdx + 1, newValue);
  }

  /**
   * Thuật toán chặn tính năng mặc định để triển khai quy tắc Shift + Click chọn dải ô đa điểm
   */
  function handleCellClickEngine(e, rIdx, cIdx) {
    if (e.shiftKey && selectionAnchor) {
      e.preventDefault();
      clearTextSelectionSelection();
      calculateSelectedRangeBounds(selectionAnchor.r, selectionAnchor.c, rIdx, cIdx);
    } else {
      selectionAnchor = { r: rIdx, c: cIdx };
      calculateSelectedRangeBounds(rIdx, cIdx, rIdx, cIdx);
    }
    highlightSelectedVisualCells();
  }

  function calculateSelectedRangeBounds(r1, c1, r2, c2) {
    selectedRangeBounds = {
      minR: Math.min(r1, r2),
      maxR: Math.max(r1, r2),
      minC: Math.min(c1, c2),
      maxC: Math.max(c1, c2)
    };
    isSelectingFullRows = false;
    isSelectingFullCols = false;
  }

  function executeFullRowSelection(e, rIdx) {
    e.preventDefault();
    var colCount = rawSpreadsheetCache[currentActiveTab][0].length;
    if (e.shiftKey && selectionAnchor !== null) {
      clearTextSelectionSelection();
      selectedRangeBounds = {
        minR: Math.min(selectionAnchor.r, rIdx), maxR: Math.max(selectionAnchor.r, rIdx),
        minC: 0, maxC: colCount - 1
      };
    } else {
      selectionAnchor = { r: rIdx, c: 0 };
      selectedRangeBounds = { minR: rIdx, maxR: rIdx, minC: 0, maxC: colCount - 1 };
    }
    isSelectingFullRows = true; isSelectingFullCols = false;
    highlightSelectedVisualCells();
  }

  function executeFullColumnSelection(e, colIdx) {
    e.preventDefault();
    var rowCount = rawSpreadsheetCache[currentActiveTab].length;
    if (e.shiftKey && selectionAnchor !== null) {
      clearTextSelectionSelection();
      selectedRangeBounds = {
        minR: 0, maxR: rowCount - 1,
        minC: Math.min(selectionAnchor.c, colIdx), maxC: Math.max(selectionAnchor.c, colIdx)
      };
    } else {
      selectionAnchor = { r: 0, c: colIdx };
      selectedRangeBounds = { minR: 0, maxR: rowCount - 1, minC: colIdx, maxC: colIdx };
    }
    isSelectingFullRows = false; isSelectingFullCols = true;
    highlightSelectedVisualCells();
  }

  function highlightSelectedVisualCells() {
    // Xóa toàn bộ viền cũ
    var cells = document.querySelectorAll(".mkt-custom-table td");
    cells.forEach(function(c) { c.classList.remove("cell-selected"); });
    
    if (!selectedRangeBounds) return;
    
    for (var r = selectedRangeBounds.minR; r <= selectedRangeBounds.maxR; r++) {
      for (var c = selectedRangeBounds.minC; c <= selectedRangeBounds.maxC; c++) {
        var cell = document.getElementById("cell_" + r + "_" + c);
        if (cell) cell.classList.add("cell-selected");
      }
    }
  }

  function clearTextSelectionSelection() {
    if (window.getSelection) { window.getSelection().removeAllRanges(); }
    else if (document.selection) { document.selection.empty(); }
  }
/**
   * Hàm tính toán thông minh giúp Menu chuột phải không bao giờ bị lọt ra ngoài màn hình
   */
  function showContextMenuSafe(menu, e) {
    menu.style.display = "block"; // Hiển thị tạm để lấy kích thước thật
    var menuWidth = menu.offsetWidth;
    var menuHeight = menu.offsetHeight;
    
    var leftPos = e.clientX;
    var topPos = e.clientY;
    
    // Nếu menu bị tràn lề phải
    if (leftPos + menuWidth > window.innerWidth) {
      leftPos = window.innerWidth - menuWidth - 5;
    }
    // Nếu menu bị tràn lề dưới (đáy màn hình)
    if (topPos + menuHeight > window.innerHeight) {
      topPos = window.innerHeight - menuHeight - 5;
    }
    
    menu.style.left = leftPos + "px";
    menu.style.top = topPos + "px";
  }

  // --- HÀM XỬ LÝ COPY / PASTE ---
  function copySelectedRangeAction() {
    if (!selectedRangeBounds) return;
    var tsvData = "";
    var dataMatrix = rawSpreadsheetCache[currentActiveTab];
    for (var r = selectedRangeBounds.minR; r <= selectedRangeBounds.maxR; r++) {
      var rowStr = [];
      for (var c = selectedRangeBounds.minC; c <= selectedRangeBounds.maxC; c++) {
        rowStr.push(dataMatrix[r][c] || "");
      }
      tsvData += rowStr.join("\t") + "\n";
    }
    navigator.clipboard.writeText(tsvData).catch(function(err) {
      console.error("Lỗi khi Copy: ", err);
    });
  }

  function pasteToSelectedRangeAction() {
    if (!selectedRangeBounds) return;
    navigator.clipboard.readText().then(function(clipText) {
      if(!clipText) return;
      var rows = clipText.split("\n");
      var startR = selectedRangeBounds.minR;
      var startC = selectedRangeBounds.minC;
      var dataMatrix = rawSpreadsheetCache[currentActiveTab];
      
      for (var i = 0; i < rows.length; i++) {
        if (!rows[i] && i === rows.length - 1) continue; // Bỏ qua dòng rỗng cuối
        var cols = rows[i].split("\t");
        var targetR = startR + i;
        if (targetR >= dataMatrix.length) {
            // Tự động thêm hàng nếu dán vượt ranh giới
            dataMatrix.push(Array(dataMatrix[0].length).fill(""));
            google.script.run.structuralChangeBackground(activeSpreadsheetId, currentActiveTab, "insertRow", targetR + 1, 1);
        }
        for (var j = 0; j < cols.length; j++) {
          var targetC = startC + j;
          if (targetC < dataMatrix[0].length) {
            dataMatrix[targetR][targetC] = cols[j];
            google.script.run.syncCellValuesBackground(activeSpreadsheetId, currentActiveTab, targetR + 1, targetC + 1, cols[j]);
          }
        }
      }
      buildSpreadsheetGrid();
    }).catch(function(err) {
      console.error("Lỗi khi Paste: (Trình duyệt có thể chặn quyền đọc Clipboard) ", err);
    });
  }

  function executeRangeColorApplication(targetMode, hexColor) {
    attributesCache[currentActiveTab] = attributesCache[currentActiveTab] || {};
    
    if (isSelectingFullRows) {
      attributesCache[currentActiveTab].rowStyles = attributesCache[currentActiveTab].rowStyles || {};
      for (var r = selectedRangeBounds.minR; r <= selectedRangeBounds.maxR; r++) {
        attributesCache[currentActiveTab].rowStyles[r] = attributesCache[currentActiveTab].rowStyles[r] || {};
        if (targetMode === 'bg') attributesCache[currentActiveTab].rowStyles[r].bg = hexColor;
        if (targetMode === 'text') attributesCache[currentActiveTab].rowStyles[r].color = hexColor;
      }
    } else if (isSelectingFullCols) {
      attributesCache[currentActiveTab].colStyles = attributesCache[currentActiveTab].colStyles || {};
      for (var c = selectedRangeBounds.minC; c <= selectedRangeBounds.maxC; c++) {
        attributesCache[currentActiveTab].colStyles[c] = attributesCache[currentActiveTab].colStyles[c] || {};
        if (targetMode === 'bg') attributesCache[currentActiveTab].colStyles[c].bg = hexColor;
        if (targetMode === 'text') attributesCache[currentActiveTab].colStyles[c].color = hexColor;
      }
    } else {
      attributesCache[currentActiveTab].cellStyles = attributesCache[currentActiveTab].cellStyles || {};
      for (var r = selectedRangeBounds.minR; r <= selectedRangeBounds.maxR; r++) {
        for (var c = selectedRangeBounds.minC; c <= selectedRangeBounds.maxC; c++) {
          var key = r + "_" + c;
          attributesCache[currentActiveTab].cellStyles[key] = attributesCache[currentActiveTab].cellStyles[key] || {};
          if (targetMode === 'bg') attributesCache[currentActiveTab].cellStyles[key].bg = hexColor;
          if (targetMode === 'text') attributesCache[currentActiveTab].cellStyles[key].color = hexColor;
        }
      }
    }
    
    buildSpreadsheetGrid();
    google.script.run.saveSheetAttributesBackground(activeSpreadsheetId, currentActiveTab, attributesCache[currentActiveTab]);
  }

  // HÀM THỰC THI DI CHUYỂN HÀNG/CỘT SAU KHI KÉO THẢ
  // HÀM THỰC THI DI CHUYỂN HÀNG KÈM THEO THUỘC TÍNH
  function executeMoveRowsAction(startR, endR, targetR) {
    if (targetR >= startR && targetR <= endR) return;
    var dataMatrix = rawSpreadsheetCache[currentActiveTab];
    var count = endR - startR + 1;
    var insertPos = targetR > endR ? targetR - count : targetR;

    // Di chuyển dữ liệu
    var extracted = dataMatrix.splice(startR, count);
    dataMatrix.splice.apply(dataMatrix, [insertPos, 0].concat(extracted));

    // Di chuyển thuộc tính cấu trúc (Màu hàng, độ cao...)
    var attrs = attributesCache[currentActiveTab] || {};
    var rowProps = ['rowHeights', 'rowStyles'];
    rowProps.forEach(function(prop) {
        if(attrs[prop]) {
            var arr = [];
            for(var i=0; i < dataMatrix.length + count; i++) arr.push(attrs[prop][i]);
            var ext = arr.splice(startR, count);
            arr.splice.apply(arr, [insertPos, 0].concat(ext));
            var newObj = {};
            arr.forEach(function(v, i) { if(v !== undefined) newObj[i] = v; });
            attrs[prop] = newObj;
        }
    });

    // Di chuyển các thuộc tính màu Ô
    if(attrs.cellStyles) {
        var newCellStyles = {};
        for(var key in attrs.cellStyles) {
            var parts = key.split('_');
            var r = parseInt(parts[0]);
            var c = parseInt(parts[1]);
            var newR = r;
            if(r >= startR && r <= endR) newR = insertPos + (r - startR);
            else if (targetR > endR && r > endR && r < targetR) newR = r - count;
            else if (targetR < startR && r >= targetR && r < startR) newR = r + count;
            newCellStyles[newR + "_" + c] = attrs.cellStyles[key];
        }
        attrs.cellStyles = newCellStyles;
    }

    buildSpreadsheetGrid();
    google.script.run.moveStructuralBackground(activeSpreadsheetId, currentActiveTab, 'row', startR + 1, endR + 1, targetR + 1);
    google.script.run.saveSheetAttributesBackground(activeSpreadsheetId, currentActiveTab, attrs);
  }

  // HÀM THỰC THI DI CHUYỂN CỘT KÈM THEO THUỘC TÍNH VÀ LOẠI CỘT
  function executeMoveColsAction(startC, endC, targetC) {
    if (targetC >= startC && targetC <= endC) return;
    var dataMatrix = rawSpreadsheetCache[currentActiveTab];
    var count = endC - startC + 1;
    var insertPos = targetC > endC ? targetC - count : targetC;

    // Di chuyển dữ liệu
    dataMatrix.forEach(function(row) {
        var extracted = row.splice(startC, count);
        row.splice.apply(row, [insertPos, 0].concat(extracted));
    });

    // Di chuyển cấu trúc (Loại Cột, Dropdown, Màu Cột...)
    var attrs = attributesCache[currentActiveTab] || {};
    var colProps = ['colWidths', 'colStyles', 'columnTypes', 'dropdownChoices'];
    colProps.forEach(function(prop) {
        if(attrs[prop]) {
            var arr = [];
            for(var i=0; i < dataMatrix[0].length + count; i++) arr.push(attrs[prop][i]);
            var ext = arr.splice(startC, count);
            arr.splice.apply(arr, [insertPos, 0].concat(ext));
            var newObj = {};
            arr.forEach(function(v, i) { if(v !== undefined) newObj[i] = v; });
            attrs[prop] = newObj;
        }
    });

    // Di chuyển thuộc tính màu Ô
    if(attrs.cellStyles) {
        var newCellStyles = {};
        for(var key in attrs.cellStyles) {
            var parts = key.split('_');
            var r = parseInt(parts[0]);
            var c = parseInt(parts[1]);
            var newC = c;
            if(c >= startC && c <= endC) newC = insertPos + (c - startC);
            else if (targetC > endC && c > endC && c < targetC) newC = c - count;
            else if (targetC < startC && c >= targetC && c < startC) newC = c + count;
            newCellStyles[r + "_" + newC] = attrs.cellStyles[key];
        }
        attrs.cellStyles = newCellStyles;
    }

    buildSpreadsheetGrid();
    google.script.run.moveStructuralBackground(activeSpreadsheetId, currentActiveTab, 'col', startC + 1, endC + 1, targetC + 1);
    google.script.run.saveSheetAttributesBackground(activeSpreadsheetId, currentActiveTab, attrs);
  }

    if(cols !== null) attributesCache[currentActiveTab].frozenCols = cols;

    google.script.run.freezePanesBackground(activeSpreadsheetId, currentActiveTab, rows, cols);
    google.script.run.saveSheetAttributesBackground(activeSpreadsheetId, currentActiveTab, attributesCache[currentActiveTab]);
  }

  function applyColFilter(colIdx) {

      cbs.forEach(function(cb) {

      buildSpreadsheetGrid(); // Tải lại bảng với dữ liệu đã lọc
  }

  function generateUniqueWebID(existingSet) {
      var chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
      var id = '';
      do {
          var str = '';
          for (var i = 0; i < 3; i++) str += chars.charAt(Math.floor(Math.random() * chars.length));
          var num = Math.floor(Math.random() * 100000).toString().padStart(5, '0');
          id = 'WEB_' + str + num;
      } while (existingSet.has(id));
      existingSet.add(id); // Thêm vào set để vòng lặp sau không bị trùng
      return id;
  }

function startSpiderSyncEngine() {
  setInterval(function() {
    // TỐI THƯỢNG: Tránh ghi đè giao diện nếu (1) Queue đang mắc kẹt chưa đẩy xong, hoặc (2) Người dùng vừa chạm vào màn hình trong vòng 3 giây qua
    if (actionQueue.length > 0 || isProcessingQueue) return;
    if (new Date().getTime() - lastUserInteractionTime < 3000) return;

    google.script.run.withSuccessHandler(function(deltaResponse) {
      if (deltaResponse.hasChanges) {
        
        // NẾU CÓ THAY ĐỔI TỪ THIẾT BỊ KHÁC: Tải lại Data mới dạng ngầm
        google.script.run.withSuccessHandler(function(newSysData) {
            if (newSysData.success) {
                // CHỐNG GHI ĐÈ KÉP LẦN CUỐI: Lỡ trong lúc đợi mạng kéo dữ liệu về (1-2s), người dùng lại vừa click/gõ phím thì hủy việc vẽ lại bảng
                if (actionQueue.length > 0 || isProcessingQueue || (new Date().getTime() - lastUserInteractionTime < 3000)) return;

                // CẬP NHẬT NGAY VÀO TRÍ NHỚ TRÌNH DUYỆT
                localStorage.setItem('MKT_OFFLINE_CACHE_' + activeSpreadsheetId, JSON.stringify(newSysData.data));

                rawSpreadsheetCache = newSysData.data.sheets;
                attributesCache = newSysData.data.attributes;
                buildSpreadsheetGrid(); 
                currentSyncToken = deltaResponse.newSyncToken;
            }
        }).getInitialSystemData(activeSpreadsheetId);
        
      }
    }).checkDeltaChangesBackground(activeSpreadsheetId, currentSyncToken);
  }, 5000); 
}

  document.addEventListener('paste', function(e) {
    // 1. Thả cửa cho trình duyệt tự xử lý nếu người dùng đang chủ động gõ chữ bên trong 1 ô (contentEditable) hoặc đang gõ vào ô tìm kiếm
    if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA' || e.target.isContentEditable) {
      return; 
    }

    // 2. Nếu chưa bôi đen vùng nào trên lưới, bỏ qua thao tác dán
    if (!selectedRangeBounds) return;

    // 3. Đóng băng hành vi dán gom cục mặc định của trình duyệt
    e.preventDefault(); 

    // 4. Móc dữ liệu từ Clipboard (Lách được rào cản bảo mật của Iframe)
    var clipText = (e.clipboardData || window.clipboardData).getData('text');
    if (!clipText) return;

    // 5. Tái sử dụng thuật toán phân tách ma trận
    var rows = clipText.split("\n");
    var startR = selectedRangeBounds.minR;
    var startC = selectedRangeBounds.minC;
    var dataMatrix = rawSpreadsheetCache[currentActiveTab];
    
    for (var i = 0; i < rows.length; i++) {
      if (!rows[i] && i === rows.length - 1) continue; // Bỏ qua dòng rỗng cuối cùng khi copy từ Excel
      var cols = rows[i].split("\t");
      
      var targetR = startR + i;
      // Tự động cơi nới thêm hàng nếu dữ liệu dán vượt quá đáy bảng
      if (targetR >= dataMatrix.length) {
          dataMatrix.push(Array(dataMatrix[0].length).fill(""));
          google.script.run.structuralChangeBackground(activeSpreadsheetId, currentActiveTab, "insertRow", targetR + 1, 1);
      }
      
      for (var j = 0; j < cols.length; j++) {
          var targetC = startC + j;
          // Chỉ dán vào nếu chưa vượt quá số lượng cột của bảng
          if (targetC < dataMatrix[0].length) {
              // Tận dụng Action Queue bạn đã viết để đẩy dữ liệu lên server mà không làm nghẽn mạng
              updateCellCacheAndSyncBackground(targetR, targetC, cols[j]);
          }
      }
    }
    
    // 6. Vẽ lại giao diện lập tức
    buildSpreadsheetGrid();
  });
