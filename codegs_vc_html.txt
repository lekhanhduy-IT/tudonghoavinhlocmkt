const SPREADSHEET_ID = "1JpGA6elt651ZIa3w3kQVyCVfbGzAcRjIScPQoZjNXt4";
const POST_IMAGE_ARCHIVE_FOLDER_ID = "11ad-tiEWqSeJdm0kv_q-_b2x0qWD9b6b";

function doGet(e) {
  var action = (e && e.parameter) ? e.parameter.action : '';

  // 1. Lấy danh sách admin
  if (action === 'get_admins') {
    var props = PropertiesService.getScriptProperties();
    var adminStr = props.getProperty('SYS_ADMIN_EMAILS');
    var admins = adminStr ? JSON.parse(adminStr) : ["admin@tudonghoavinhloc.com"]; 
    return ContentService.createTextOutput(JSON.stringify({ success: true, admins: admins })).setMimeType(ContentService.MimeType.JSON);
  }

  // 2. LẤY DANH SÁCH PROJECT (Các folder con trong Shared Drive)
  if (action === 'get_projects') {
    var parentId = e.parameter.parent || "11ad-tiEWqSeJdm0kv_q-_b2x0qWD9b6b";
    return ContentService.createTextOutput(JSON.stringify(getDriveProjects_(parentId))).setMimeType(ContentService.MimeType.JSON);
  }

  // 3. LẤY DANH SÁCH ẢNH TRONG 1 PROJECT (Folder con)
  if (action === 'get_images') {
    var folderId = e.parameter.folderId;
    return ContentService.createTextOutput(JSON.stringify(getFolderImages_(folderId))).setMimeType(ContentService.MimeType.JSON);
  }
// LẤY TÀI NGUYÊN TỪ CANVA_HIDDEN VÀ SHARED DRIVE
  if (action === 'get_canva_resources') {
    var resFolderId = "1JOMYxGCnoj4nhVaV37EDwPuq4dKcUiCL?usp=sharing".match(/folders\/([a-zA-Z0-9_-]+)/)[1];
    var images = getFolderImages_(resFolderId).images || [];
    
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName("Canva_hidden");
    var shapes = [], frames = [];
    if (sheet) {
      var vals = sheet.getDataRange().getValues();
      for (var i = 1; i < vals.length; i++) {
        if (vals[i][0] === "Shape") shapes.push({ code: vals[i][1] });
        if (vals[i][0] === "Frame") frames.push({ code: vals[i][1] });
      }
    }
    return ContentService.createTextOutput(JSON.stringify({
      success: true, images: images, shapes: shapes, frames: frames
    })).setMimeType(ContentService.MimeType.JSON);
  }
  // Mặc định: Trả về dữ liệu Sheet
  var data = getFullSystemData();
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

// --- HÀM QUÉ́T DANH SÁCH FOLDER CON ---
function getDriveProjects_(parentId) {
  try {
    var parentFolder = DriveApp.getFolderById(parentId);
    var subFolders = parentFolder.getFolders();
    var folders = [];
    while (subFolders.hasNext()) {
      var f = subFolders.next();
      folders.push({ id: f.getId(), name: f.getName() });
    }
    return { success: true, folders: folders };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

// --- HÀM QUÉT TOÀN BỘ FILE ẢNH TRONG FOLDER ---
function getFolderImages_(folderId) {
  try {
    var folder = DriveApp.getFolderById(folderId);
    var files = folder.getFiles();
    var images = [];
    while (files.hasNext()) {
      var file = files.next();
      var mime = file.getMimeType();
      // Chỉ lấy các file có định dạng hình ảnh
      if (mime.indexOf('image/') === 0) {
        images.push({
          id: file.getId(),
          name: file.getName(),
          mimeType: mime
        });
      }
    }
    return { success: true, images: images };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents || "{}");
    var result;

    // Payload dạng mảng là hàng đợi đồng bộ Sheet cũ.
    if (Array.isArray(payload)) {
      result = processActionQueue(payload);
    }
    // Payload dạng object dành cho các API riêng.
    else if (payload && payload.action === "UPLOAD_POST_IMAGES") {
      result = uploadPostImagesToSharedDrive(payload);
    }
    else {
      result = { success: false, error: "Unsupported action" };
    }

    return ContentService
      .createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({success: false, error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function uploadPostImagesToSharedDrive(payload) {
  var images = payload.images || [];
  if (!images.length) return { success: true, files: [] };

  var rootFolder = DriveApp.getFolderById(POST_IMAGE_ARCHIVE_FOLDER_ID);
  var postId = sanitizeDriveName_(payload.postId || "BAI_VIET_KHONG_ID");

  // Tránh hai thiết bị đồng thời tạo hai thư mục con cùng tên.
  var lock = LockService.getScriptLock();
  lock.waitLock(30000);
  var postFolder;
  try {
    postFolder = getOrCreateSubFolder_(rootFolder, postId);
  } finally {
    lock.releaseLock();
  }

  var output = [];

  images.forEach(function(item, index) {
    try {
      var metadata = item.metadata || {};
      var fileName = sanitizeFileName_(item.fileName || metadata.fileName || (postId + "-image-" + (index + 1)));
      var source = String(item.source || "");
      var blob = buildImageBlobFromSource_(source, fileName, item.mimeType || "");
      var finalName = ensureImageExtension_(fileName, blob.getContentType());
      blob.setName(finalName);

      var file = postFolder.createFile(blob);
      file.setName(finalName);

      // Metadata được lưu thêm vào Description của Drive để dự phòng độc lập với Sheet.
      try {
        file.setDescription(JSON.stringify({
          postId: postId,
          imageId: item.clientId || "",
          role: item.role || "content",
          metadata: metadata,
          savedAt: new Date().toISOString()
        }));
      } catch (descErr) {}

      // Shared Drive thường kế thừa quyền từ thư mục. Nếu chính sách cho phép,
      // lệnh này giúp URL ảnh có thể được Vercel/CDN đọc trực tiếp.
      try {
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      } catch (sharingErr) {}

      var fileId = file.getId();
      output.push({
        success: true,
        clientId: item.clientId || "",
        fileId: fileId,
        fileName: finalName,
        driveUrl: "https://drive.google.com/uc?export=view&id=" + fileId,
        downloadUrl: "https://drive.google.com/uc?export=download&id=" + fileId,
        folderUrl: "https://drive.google.com/drive/folders/" + postFolder.getId()
      });
    } catch (imageErr) {
      output.push({
        success: false,
        clientId: item.clientId || "",
        error: imageErr.toString()
      });
    }
  });

  var failed = output.filter(function(x) { return !x.success; });
  return {
    success: failed.length === 0,
    files: output,
    error: failed.length ? (failed.length + " ảnh tải lên thất bại") : ""
  };
}

function getOrCreateSubFolder_(parentFolder, name) {
  var folders = parentFolder.getFoldersByName(name);
  if (folders.hasNext()) return folders.next();
  return parentFolder.createFolder(name);
}

function buildImageBlobFromSource_(source, fileName, mimeHint) {
  if (!source) throw new Error("Ảnh không có dữ liệu nguồn.");

  // Data URL từ ảnh dán hoặc tải lên.
  if (source.indexOf("data:") === 0) {
    var match = source.match(/^data:([^;,]+)?(?:;charset=[^;,]+)?;base64,(.+)$/);
    if (!match) throw new Error("Data URL ảnh không hợp lệ.");
    var mime = match[1] || mimeHint || "application/octet-stream";
    var bytes = Utilities.base64Decode(match[2]);
    return Utilities.newBlob(bytes, mime, fileName);
  }

  // Nếu là link Google Drive và Apps Script có quyền, sao chép byte trực tiếp.
  var driveId = extractDriveFileId_(source);
  if (driveId) {
    try {
      return DriveApp.getFileById(driveId).getBlob().setName(fileName);
    } catch (driveErr) {
      // Không có quyền thì rơi xuống UrlFetchApp.
    }
  }

  if (!/^https?:\/\//i.test(source)) {
    throw new Error("Nguồn ảnh phải là data URL hoặc URL http/https.");
  }

  var response = UrlFetchApp.fetch(source, {
    muteHttpExceptions: true,
    followRedirects: true,
    validateHttpsCertificates: true,
    headers: {
      "User-Agent": "Mozilla/5.0 VinhLocMKT-ImageArchive/1.0"
    }
  });

  var status = response.getResponseCode();
  if (status < 200 || status >= 300) {
    throw new Error("Không tải được ảnh nguồn. HTTP " + status);
  }

  var blob = response.getBlob();
  var mime = blob.getContentType() || mimeHint || "application/octet-stream";
  if (String(mime).toLowerCase().indexOf("image/") !== 0) {
    throw new Error("URL nguồn không trả về file ảnh hợp lệ (" + mime + ").");
  }
  return Utilities.newBlob(blob.getBytes(), mime, fileName);
}

function extractDriveFileId_(url) {
  var text = String(url || "");
  var patterns = [
    /\/d\/([a-zA-Z0-9_-]{10,})/,
    /[?&]id=([a-zA-Z0-9_-]{10,})/,
    /\/thumbnail\?id=([a-zA-Z0-9_-]{10,})/
  ];
  for (var i = 0; i < patterns.length; i++) {
    var m = text.match(patterns[i]);
    if (m) return m[1];
  }
  return "";
}

function sanitizeDriveName_(name) {
  return String(name || "BAI_VIET")
    .replace(/[\\\/:*?"<>|#%{}\[\]~]/g, "-")
    .replace(/\s+/g, " ")
    .trim()
    .substring(0, 120) || "BAI_VIET";
}

function sanitizeFileName_(name) {
  return String(name || "image")
    .replace(/[\\\/:*?"<>|#%{}\[\]~]/g, "-")
    .replace(/\s+/g, "-")
    .replace(/-+/g, "-")
    .replace(/^\.+/, "")
    .trim()
    .substring(0, 180) || "image";
}

function ensureImageExtension_(fileName, mimeType) {
  if (/\.[a-zA-Z0-9]{2,5}$/.test(fileName)) return fileName;
  var map = {
    "image/jpeg": ".jpg",
    "image/png": ".png",
    "image/webp": ".webp",
    "image/gif": ".gif",
    "image/svg+xml": ".svg",
    "image/avif": ".avif"
  };
  return fileName + (map[mimeType] || ".jpg");
}

function getFullSystemData() {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var sheets = ss.getSheets();
  var result = { sheets: {}, attributes: {}, tabOrder: [] };

  sheets.forEach(function(sheet) {
    var name = sheet.getName();
    if (name === "_SYS_CONFIG_") return; 

    if (name.indexOf("_attributes") !== -1) {
      var baseName = name.replace("_attributes", "");
      var attrContent = sheet.getRange("A1").getValue();
      if (attrContent) { try { result.attributes[baseName] = JSON.parse(attrContent); } catch(e){} }
      return; 
    }

    var data = sheet.getDataRange().getDisplayValues();
    if (data.length === 0) data = [[""]];
    result.sheets[name] = data;
    result.tabOrder.push(name);
  });
  
  // Lấy Token thời gian cập nhật mới nhất
  var props = PropertiesService.getScriptProperties();
  var lastToken = props.getProperty('SYNC_TOKEN_' + SPREADSHEET_ID) || new Date().getTime().toString();
  result.syncToken = parseInt(lastToken);

  return { success: true, data: result };
}

function processActionQueue(batch) {
  var hasChanges = false;

  if (batch && batch.length > 0) {
    hasChanges = true;
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    
    batch.forEach(function(action) {
      var sheet = ss.getSheetByName(action.tab);
      // Cấp phép cho lệnh MANAGE_ADMINS đi qua mà không cần tên Tab
      if (!sheet && action.type !== 'MANAGE_TAB' && action.type !== 'MANAGE_ADMINS') return;

      if (action.type === 'SYNC_CELL') {
        var targetRange = sheet.getRange(action.row, action.col);
        if (action.val === true || action.val === "true") targetRange.insertCheckboxes().check();
        else if (action.val === false || action.val === "false") targetRange.insertCheckboxes().uncheck();
        else targetRange.setValue(action.val);
      } 
      else if (action.type === 'STRUCTURAL') {
        if (action.action === "insertRow") sheet.insertRows(action.index, action.count);
        if (action.action === "insertCol") sheet.insertColumns(action.index, action.count);
        if (action.action === "deleteRow") sheet.deleteRows(action.index, action.count);
        if (action.action === "deleteCol") sheet.deleteColumns(action.index, action.count);
      } 
      else if (action.type === 'MOVE') {
        var count = action.end - action.start + 1;
        if (action.dim === 'row') {
          sheet.moveRows(sheet.getRange(action.start, 1, count, sheet.getMaxColumns()), action.target);
        } else {
          sheet.moveColumns(sheet.getRange(1, action.start, sheet.getMaxRows(), count), action.target);
        }
      } 
      else if (action.type === 'FREEZE') {
        if(action.rows !== null) sheet.setFrozenRows(action.rows);
        if(action.cols !== null) sheet.setFrozenColumns(action.cols);
      } 
      else if (action.type === 'MANAGE_TAB') {
        if (action.action === "create") ss.insertSheet(action.tab, ss.getSheets().length);
        else if (action.action === "rename") { 
          var sh = ss.getSheetByName(action.oldName); if(sh) sh.setName(action.newName);
          var attrSh = ss.getSheetByName(action.oldName + "_attributes"); if(attrSh) attrSh.setName(action.newName + "_attributes");
        }
        else if (action.action === "duplicate") { 
          var sh = ss.getSheetByName(action.oldName); if(sh) { var ns = sh.copyTo(ss); ns.setName(action.newName); } 
          var attrSh = ss.getSheetByName(action.oldName + "_attributes"); if(attrSh) { var nas = attrSh.copyTo(ss); nas.setName(action.newName + "_attributes"); nas.hideSheet(); }
        }
        else if (action.action === "delete") { 
          var sh = ss.getSheetByName(action.tab); if(sh) ss.deleteSheet(sh); 
          var attrSh = ss.getSheetByName(action.tab + "_attributes"); if(attrSh) ss.deleteSheet(attrSh);
        }
      }
      else if (action.type === 'MANAGE_ADMINS') {
        var props = PropertiesService.getScriptProperties();
        // Đảm bảo Master Admin luôn tồn tại để tránh tự khóa bản thân
        var safeEmails = action.emails;
        if (safeEmails.indexOf("admin@tudonghoavinhloc.com") === -1) {
            safeEmails.unshift("admin@tudonghoavinhloc.com");
        }
        props.setProperty('SYS_ADMIN_EMAILS', JSON.stringify(safeEmails));
      }
      else if (action.type === 'SAVE_TAB_ATTRS') {
        var attrTabName = action.targetTab + "_attributes";
        var configSheet = ss.getSheetByName(attrTabName);
        if (!configSheet) { configSheet = ss.insertSheet(attrTabName); configSheet.hideSheet(); }
        configSheet.clear();
        configSheet.getRange("A1").setValue(JSON.stringify(action.attributes));
      }
      
    });

    SpreadsheetApp.flush(); 
  }

  // Cập nhật Dấu thời gian khi có thao tác
  var props = PropertiesService.getScriptProperties();
  var tokenKey = 'SYNC_TOKEN_' + SPREADSHEET_ID;
  if (hasChanges) {
    props.setProperty(tokenKey, new Date().getTime().toString());
  }
  var lastToken = props.getProperty(tokenKey) || new Date().getTime().toString();

  return { success: true, syncToken: parseInt(lastToken) };
}
